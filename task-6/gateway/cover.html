package api

import (
        "fmt"
        "time"

        "gateway/internal/ledger"
)

type CreateTransactionRequest struct {
        Amount      float64 `json:"amount"`
        Category    string  `json:"category"`
        Description string  `json:"description"`
        Date        string  `json:"date"`
}

type TransactionResponse struct {
        ID          int     `json:"id"`
        Amount      float64 `json:"amount"`
        Category    string  `json:"category"`
        Description string  `json:"description"`
        Date        string  `json:"date"`
}

type CreateBudgetRequest struct {
        Category string  `json:"category"`
        Limit    float64 `json:"limit"`
}

type BudgetResponse struct {
        Category string  `json:"category"`
        Limit    float64 `json:"limit"`
        Period   string  `json:"period"`
}

func (r CreateTransactionRequest) ToDomain() (ledger.Transaction, error) {
        tx := ledger.Transaction{
                Amount:      r.Amount,
                Category:    r.Category,
                Description: r.Description,
        }
        if r.Date != "" {
                t, err := time.Parse("2006-01-02", r.Date)
                if err != nil {
                        return ledger.Transaction{}, fmt.Errorf("invalid date %q, expected format YYYY-MM-DD", r.Date)
                }
                tx.Date = t
        }
        return tx, nil
}

func FromDomainTransaction(t ledger.Transaction) TransactionResponse {
        return TransactionResponse{
                ID:          t.ID,
                Amount:      t.Amount,
                Category:    t.Category,
                Description: t.Description,
                Date:        t.Date.Format("2006-01-02"),
        }
}

func (r CreateBudgetRequest) ToDomain() ledger.Budget {
        return ledger.Budget{
                Category: r.Category,
                Limit:    r.Limit,
                Period:   "month",
        }
}

func FromDomainBudget(b ledger.Budget) BudgetResponse {
        return BudgetResponse{
                Category: b.Category,
                Limit:    b.Limit,
                Period:   b.Period,
        }
}

import (
        "encoding/json"
        "errors"
        "log"

        "gateway/internal/ledger"

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

func writeError(w http.ResponseWriter, status int, msg string) {
        writeJSON(w, status, map[string]string{"error": msg})
}

func CreateTransactionHandler(w http.ResponseWriter, r *http.Request) {
        var req CreateTransactionRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeError(w, http.StatusBadRequest, "invalid JSON")
                return
        }

        tx, err := req.ToDomain()
        if err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }

        if err := tx.Validate(); err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }

        created, err := ledger.AddTransaction(tx)
        if err != nil {
                if errors.Is(err, ledger.ErrBudgetExceeded) {
                        writeError(w, http.StatusConflict, "budget exceeded")
                        return
                }
                log.Printf("AddTransaction: %v", err)
                writeError(w, http.StatusInternalServerError, "internal server error")
                return
        }

        writeJSON(w, http.StatusCreated, FromDomainTransaction(created))
}

func ListTransactionsHandler(w http.ResponseWriter, r *http.Request) {
        txs := ledger.ListTransactions()
        resp := make([]TransactionResponse, 0, len(txs))
        for _, t := range txs {
                resp = append(resp, FromDomainTransaction(t))
        }
        writeJSON(w, http.StatusOK, resp)
}

func CreateBudgetHandler(w http.ResponseWriter, r *http.Request) {
        var req CreateBudgetRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeError(w, http.StatusBadRequest, "invalid JSON")
                return
        }

        b := req.ToDomain()
        if err := b.Validate(); err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }

        if err := ledger.SetBudget(b); err != nil {
                log.Printf("SetBudget: %v", err)
                writeError(w, http.StatusInternalServerError, "internal server error")
                return
        }

        writeJSON(w, http.StatusCreated, FromDomainBudget(b))
}

func ListBudgetsHandler(w http.ResponseWriter, r *http.Request) {
        budgets := ledger.ListBudgets()
        resp := make([]BudgetResponse, 0, len(budgets))
        for _, b := range budgets {
                resp = append(resp, FromDomainBudget(b))
        }
        writeJSON(w, http.StatusOK, resp)
}

func LoggingMiddleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                next.ServeHTTP(w, r)
                log.Printf("%s %s %v", r.Method, r.URL.Path, time.Since(start))
        })
}

package ledger

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "math"
        "strings"
        "sync"
)

type Budget struct {
        Category string
        Limit    float64
        Period   string
}

var (
        transactions = make([]Transaction, 0)
        budgets      = make(map[string]Budget)
        mu           sync.RWMutex
)

func (b Budget) Validate() error {
        if strings.TrimSpace(b.Category) == "" {
                return errors.New("budget category cannot be empty")
        }

        if math.IsNaN(b.Limit) || math.IsInf(b.Limit, 0) || b.Limit <= 0 {
                return errors.New("budget limit must be a finite positive number")
        }

        return nil
}

func SetBudget(b Budget) error {
        if err := b.Validate(); err != nil {
                return fmt.Errorf("invalid budget: %w", err)
        }

        mu.Lock()
        defer mu.Unlock()

        budgets[b.Category] = b
        return nil
}

func ListBudgets() []Budget {
        mu.RLock()
        defer mu.RUnlock()

        result := make([]Budget, 0, len(budgets))
        for _, b := range budgets {
                result = append(result, b)
        }

        return result
}

func LoadBudgets(r io.Reader) error {
        var loaded []Budget

        if err := json.NewDecoder(r).Decode(&loaded); err != nil {
                return fmt.Errorf("failed to parse budgets JSON: %w", err)
        }

        for i, b := range loaded {
                if err := b.Validate(); err != nil {
                        return fmt.Errorf("invalid budget at index %d: %w", i, err)
                }
        }

        mu.Lock()
        defer mu.Unlock()

        for _, b := range loaded {
                budgets[b.Category] = b
        }

        return nil
}

package ledger

import (
        "errors"
        "fmt"
        "math"
        "strings"
        "time"
)

type Transaction struct {
        ID          int
        Amount      float64
        Category    string
        Description string
        Date        time.Time
}

func (tx Transaction) Validate() error {
        if math.IsNaN(tx.Amount) || math.IsInf(tx.Amount, 0) || tx.Amount <= 0 {
                return errors.New("transaction amount must be a finite positive number")
        }

        if strings.TrimSpace(tx.Category) == "" {
                return errors.New("transaction category cannot be empty")
        }

        if tx.Date.IsZero() {
                return errors.New("transaction date cannot be empty")
        }

        return nil
}

var ErrBudgetExceeded = errors.New("budget exceeded")

func AddTransaction(tx Transaction) (Transaction, error) {
        if tx.Date.IsZero() {
                tx.Date = time.Now()
        }

        if err := tx.Validate(); err != nil {
                return Transaction{}, fmt.Errorf("invalid transaction: %w", err)
        }

        mu.Lock()
        defer mu.Unlock()

        if b, ok := budgets[tx.Category]; ok {
                var total float64
                for _, existing := range transactions {
                        if existing.Category == tx.Category {
                                total += existing.Amount
                        }
                }
                if total+tx.Amount > b.Limit {
                        return Transaction{}, fmt.Errorf(
                                "%w for %q: total %.2f + amount %.2f > limit %.2f",
                                ErrBudgetExceeded, tx.Category, total, tx.Amount, b.Limit,
                        )
                }
        }

        tx.ID = len(transactions) + 1
        transactions = append(transactions, tx)

        return tx, nil
}

func ListTransactions() []Transaction {
        mu.RLock()
        defer mu.RUnlock()

        result := make([]Transaction, len(transactions))
        copy(result, transactions)

        return result
}

func Reset() {
        mu.Lock()
        defer mu.Unlock()
        transactions = make([]Transaction, 0)
        budgets = make(map[string]Budget)
}

const serverAddress = ":8080"

func main() {
        for _, b := range []ledger.Budget{
                {Category: "Продукты", Limit: 5000, Period: "month"},
                {Category: "Транспорт", Limit: 1000, Period: "month"},
        } {
                if err := ledger.SetBudget(b); err != nil {
                        log.Fatalf("failed to set default budget: %v", err)
                }
        }

        mux := http.NewServeMux()
        mux.HandleFunc("POST /api/transactions", api.CreateTransactionHandler)
        mux.HandleFunc("GET /api/transactions", api.ListTransactionsHandler)
        mux.HandleFunc("POST /api/budgets", api.CreateBudgetHandler)
        mux.HandleFunc("GET /api/budgets", api.ListBudgetsHandler)

        handler := api.LoggingMiddleware(mux)

        log.Printf("Gateway service started on http://localhost%s", serverAddress)
        if err := http.ListenAndServe(serverAddress, handler); err != nil {
                log.Fatalf("gateway server failed: %v", err)
        }
}
